cmake_minimum_required(VERSION 3.16)
project(deadline_lab VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(DL_SANITIZE "Address and undefined-behaviour sanitizers for local checks" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(DL_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(dl-core STATIC src/core.c)
target_include_directories(dl-core PUBLIC include)
add_executable(dl-worker src/worker_posix.c)
target_link_libraries(dl-worker PRIVATE dl-core)
add_executable(dl-core-tests tests/test_core.c)
target_link_libraries(dl-core-tests PRIVATE dl-core)
enable_testing()
add_test(NAME core-contract COMMAND dl-core-tests)
if(CMAKE_SYSTEM_NAME STREQUAL "QNX")
  add_executable(dl-qnx-server qnx/server.c)
  target_link_libraries(dl-qnx-server PRIVATE dl-core)
  target_include_directories(dl-qnx-server PRIVATE qnx)
  add_executable(dl-qnx-client qnx/client.c)
  target_include_directories(dl-qnx-client PRIVATE include qnx)
endif()
