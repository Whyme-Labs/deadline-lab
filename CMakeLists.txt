cmake_minimum_required(VERSION 3.16)
project(deadline_lab VERSION 0.6.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(DL_SANITIZE "Address and undefined-behaviour sanitizers for local checks" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(DL_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(dl-core STATIC src/core.c src/effect.c)
target_include_directories(dl-core PUBLIC include)
add_executable(dl-worker src/worker_posix.c)
target_link_libraries(dl-worker PRIVATE dl-core)
add_executable(dl-core-tests tests/test_core.c)
target_link_libraries(dl-core-tests PRIVATE dl-core)
add_executable(dl-effect-tests tests/test_effect.c)
target_link_libraries(dl-effect-tests PRIVATE dl-core)
add_library(dl-sha256 STATIC src/sha256.c)
target_include_directories(dl-sha256 PUBLIC include)
add_executable(dl-capture src/capture.c)
target_link_libraries(dl-capture PRIVATE dl-sha256)
add_executable(dl-sha256-tests tests/test_sha256.c)
target_link_libraries(dl-sha256-tests PRIVATE dl-sha256)
enable_testing()
add_test(NAME artifact-digest COMMAND dl-sha256-tests)
add_test(NAME core-contract COMMAND dl-core-tests)
add_test(NAME effect-sink COMMAND dl-effect-tests)
if(CMAKE_SYSTEM_NAME STREQUAL "QNX")
  add_executable(dl-qnx-abi qnx/abi_probe.c)
  target_include_directories(dl-qnx-abi PRIVATE include qnx)
  add_executable(dl-qnx-server qnx/server.c)
  target_link_libraries(dl-qnx-server PRIVATE dl-core)
  target_include_directories(dl-qnx-server PRIVATE qnx)
  add_executable(dl-qnx-client qnx/client.c)
  target_include_directories(dl-qnx-client PRIVATE include qnx)
endif()

if(NOT CMAKE_SYSTEM_NAME STREQUAL "QNX")
  foreach(role server client abi)
    add_executable(dl-transport-${role} tests/transport_fixture.c)
  endforeach()
endif()
